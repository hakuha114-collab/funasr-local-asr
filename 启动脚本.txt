# FunASR Runtime SDK 在线 2pass 服务启动脚本（2026-08-26 更新）
# 更新点:
#   1. 镜像 0.1.12 -> 0.1.13（官方最新在线 CPU 镜像，含若干 bug 修复）
#   2. 新增说话人分离: --spk-dir 加载 cam++ 模型（容器首次启动会自动下载到 ./models）
#   3. hotword 文件缺失会导致启动报错，已在容器内创建空文件
#   4. 端口说明: 容器内 10095，宿主机映射 10096；客户端连宿主机请用 10096

# 1) 拉取最新镜像
docker pull registry.cn-hangzhou.aliyuncs.com/funasr_repo/funasr:funasr-runtime-sdk-online-cpu-0.1.13

# 2) 准备模型目录（Windows 用 mkdir，Git Bash 可直接执行）
mkdir -p ./funasr-runtime-resources/models

# 3) 启动容器（宿主机 10096 -> 容器 10095）
cd F:\ai\funasr_repo
docker run -p 10096:10095 -it --privileged=true -v models:/workspace/models registry.cn-hangzhou.aliyuncs.com/funasr_repo/funasr:funasr-runtime-sdk-online-cpu-0.1.13
